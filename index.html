let selectedVoice = "Zephyr"; let selectedVoiceName = "Zephyr"; let audioUrl = null; // ======================================== // Text ကို သဘာဝကျကျ အပိုင်းခွဲခြင်း // ======================================== function splitText(text, maxLength = 1000) { const cleanText = text.trim(); let bestCut = -1; for (const mark of punctuationMarks) { const position = remaining.lastIndexOf(mark, maxLength); if (position > bestCut) { bestCut = position; } } // စာကြောင်းဆုံးရာ မတွေ့ရင် space နေရာမှာ ဖြတ် if (bestCut < Math.floor(maxLength * 0.6)) { const spaceCut = remaining.lastIndexOf(" ", maxLength); } if (remaining.length > 0) { chunks.push(remaining); } return chunks; } // ======================================== // WAV Audio တွေကို ပြန်ပေါင်းခြင်း // ======================================== function mergeWavFiles(buffers) { if (!buffers.length) { throw new Error("Audio file မရှိပါ။"); } const result = new Uint8Array( 44 + totalDataLength ); // Header copy result.set(header, 0); let offset = 44; for (const buffer of buffers) { const data = new Uint8Array(buffer); result.set( data.slice(44), offset ); offset += data.length - 44; } const view = new DataView( result.buffer ); // RIFF file size view.setUint32( 4, 36 + totalDataLength, true ); // WAV data size view.setUint32( 40, totalDataLength, true ); return result.buffer; } // ======================================== // Generate Voice // ======================================== async function generateVoice() { const text = document .getElementById("text") .value .trim();
